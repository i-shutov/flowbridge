cmake_minimum_required(VERSION 3.24)
project(FlowBridgeNative VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This target requires Windows or a Windows cross toolchain.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FLOWBRIDGE_FFI_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release/flowbridge_ffi.lib" CACHE FILEPATH "Rust FFI import library or static library")
if(NOT EXISTS "${FLOWBRIDGE_FFI_LIBRARY}")
  message(FATAL_ERROR "Build flowbridge-ffi, then set FLOWBRIDGE_FFI_LIBRARY to its .lib or .dll.a path.")
endif()
add_library(flowbridge_ffi UNKNOWN IMPORTED)
set_target_properties(flowbridge_ffi PROPERTIES IMPORTED_LOCATION "${FLOWBRIDGE_FFI_LIBRARY}")
find_package(Threads REQUIRED)
add_executable(FlowBridge WIN32 main.cpp)
target_compile_definitions(FlowBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_include_directories(FlowBridge PRIVATE ../include ../../third_party)
target_link_libraries(FlowBridge PRIVATE flowbridge_ffi Threads::Threads comctl32 comdlg32 shell32 ole32 uuid user32 gdi32 advapi32 userenv ws2_32 bcrypt ntdll)
if(MSVC)
  target_compile_options(FlowBridge PRIVATE /W4 /utf-8)
  set_property(TARGET FlowBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
  target_compile_options(FlowBridge PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(FlowBridge PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
# The optional capture helper is packaged independently (without its kernel
# driver). The GUI elevation request explicitly requires user approval.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/wfp/redirect_io.cpp")
  add_library(flowbridge_wfp STATIC wfp/redirect_io.cpp)
  target_compile_definitions(flowbridge_wfp PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(flowbridge_wfp PUBLIC ws2_32)
endif()
install(TARGETS FlowBridge RUNTIME DESTINATION .)

option(FLOWBRIDGE_BUILD_NATIVE_TESTS "Build Win32 ABI and GUI smoke harness (disposable profile only)" OFF)
if(FLOWBRIDGE_BUILD_NATIVE_TESTS)
  add_executable(FlowBridgeRuleModelTest ../../tests/rule_model_test.cpp)
  target_include_directories(FlowBridgeRuleModelTest PRIVATE ../../third_party)
  if(MSVC)
    target_compile_options(FlowBridgeRuleModelTest PRIVATE /W4 /utf-8)
    set_property(TARGET FlowBridgeRuleModelTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW)
    target_compile_options(FlowBridgeRuleModelTest PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  # Dependency-free profile export to a separately installed Microsoft-signed
  # Windows Packet Filter / ProxiFyre sidecar. No kernel driver is packaged.
  add_executable(FlowBridgeProxiFyreExportTest ../../tests/proxifyre_export_test.cpp)
  target_include_directories(FlowBridgeProxiFyreExportTest PRIVATE ../../third_party)
  if(MSVC)
    target_compile_options(FlowBridgeProxiFyreExportTest PRIVATE /W4 /utf-8)
    set_property(TARGET FlowBridgeProxiFyreExportTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW)
    target_compile_options(FlowBridgeProxiFyreExportTest PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(FlowBridgeNativeSmoke ../../tests/windows_native_smoke.cpp)
  target_include_directories(FlowBridgeNativeSmoke PRIVATE ../include ../../third_party)
  target_link_libraries(FlowBridgeNativeSmoke PRIVATE flowbridge_ffi Threads::Threads ws2_32 iphlpapi user32 gdi32 comctl32 userenv bcrypt ntdll shell32 ole32)
  if(MSVC)
    target_compile_options(FlowBridgeNativeSmoke PRIVATE /W4 /utf-8)
    set_property(TARGET FlowBridgeNativeSmoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW)
    target_compile_options(FlowBridgeNativeSmoke PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(FlowBridgeNativeSmoke PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()
