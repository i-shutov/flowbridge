cmake_minimum_required(VERSION 3.24)
project(FlowBridgeCapture LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows user-mode service; kernel target uses driver/FlowBridgeWfp.vcxproj")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FLOWBRIDGE_FFI_LIBRARY "" CACHE FILEPATH "Windows Rust static library or DLL import library")
if(NOT EXISTS "${FLOWBRIDGE_FFI_LIBRARY}")
  message(FATAL_ERROR "Set FLOWBRIDGE_FFI_LIBRARY to a Windows flowbridge_ffi library")
endif()
find_package(Threads REQUIRED)
add_executable(FlowBridgeCapture service.cpp capture.cpp redirect_io.cpp)
target_include_directories(FlowBridgeCapture PRIVATE ../../include ../../../third_party)
target_compile_definitions(FlowBridgeCapture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(FlowBridgeCapture PRIVATE "${FLOWBRIDGE_FFI_LIBRARY}" Threads::Threads fwpuclnt ws2_32 advapi32 userenv bcrypt ntdll shell32 ole32 uuid)
if(MSVC)
  target_compile_options(FlowBridgeCapture PRIVATE /W4 /utf-8)
  set_property(TARGET FlowBridgeCapture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(FlowBridgeCapture PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(FlowBridgeCapture PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
# Intentionally no install()/driver registration. A signed, tested Windows driver
# package and reviewed service ACL/credentials deployment precede distribution.
