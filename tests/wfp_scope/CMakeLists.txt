cmake_minimum_required(VERSION 3.24)
project(FlowBridgeWfpScopeQA LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This test uses the real Windows Base Filtering Engine")
endif()
add_executable(FlowBridgeWfpScopeQA ../windows_wfp_scope.cpp)
target_compile_features(FlowBridgeWfpScopeQA PRIVATE cxx_std_17)
target_compile_definitions(FlowBridgeWfpScopeQA PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(FlowBridgeWfpScopeQA PRIVATE fwpuclnt ws2_32 rpcrt4)
if(MSVC)
  target_compile_options(FlowBridgeWfpScopeQA PRIVATE /W4 /WX /utf-8)
  set_property(TARGET FlowBridgeWfpScopeQA PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
