cmake_minimum_required(VERSION 3.24)
project(FlowBridgeGuiOwnerEpochQA LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Native Windows kernel process handle test; no emulation")
endif()
set(GUI_OWNER_EPOCH_HEADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../native/windows"
    CACHE PATH "Production or explicitly mutated GUI owner header")
add_executable(FlowBridgeGuiOwnerEpochQA ../windows_gui_owner_epoch_runtime.cpp)
target_include_directories(FlowBridgeGuiOwnerEpochQA PRIVATE "${GUI_OWNER_EPOCH_HEADER_DIR}")
target_compile_features(FlowBridgeGuiOwnerEpochQA PRIVATE cxx_std_17)
target_compile_definitions(FlowBridgeGuiOwnerEpochQA PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(FlowBridgeGuiOwnerEpochQA PRIVATE /W4 /WX /utf-8)
  set_property(TARGET FlowBridgeGuiOwnerEpochQA PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
